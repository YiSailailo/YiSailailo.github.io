贪吃蛇小游戏

发表于2026-10-01|更新于2026-10-01
|浏览量:

今天用网页写了一个贪吃蛇小游戏。以后想玩随时点进来，方向键（↑↓←→）或 WASD 控制移动，吃到红色食物就变长、得分、速度还会慢慢加快，撞墙或咬到自己就结束，按空格键重新开始。手机上可以用屏幕上的方向按钮玩。

得分：0
最高分：0
方向键 / WASD 控制 · 空格键开始/重开

下面是这个游戏的完整代码（纯 HTML，复制保存成 .html 文件就能在任何浏览器里打开玩）：

<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>贪吃蛇</title>
  <style>
    body{margin:0;min-height:100vh;display:flex;align-items:center;justify-content:center;background:#111;font-family:system-ui,sans-serif}
    .wrap{text-align:center;color:#fff}
    canvas{background:#1e272e;border:2px solid #263238;border-radius:8px}
    .row{margin-top:12px;font-size:18px}
    .pad{display:none;margin-top:12px}
    .pad button{width:60px;height:50px;font-size:22px;border:none;border-radius:8px;background:#333;color:#fff}
    @media (pointer:coarse){.pad{display:flex;flex-direction:column;align-items:center;gap:6px}}
  </style>
</head>
<body>
  <div class="wrap">
    <div class="row">得分:<span id="score">0</span>   最高:<span id="best">0</span></div>
    <canvas id="c" width="400" height="400"></canvas>
    <div class="row">方向键/WASD 移动 · 空格 开始/重开</div>
    <div class="pad">
      <div><button data-d="up">▲</button></div>
      <div><button data-d="left">◀</button><button data-d="down">▼</button><button data-d="right">▶</button></div>
    </div>
  </div>
  <script>
    const cv=document.getElementById('c'),ctx=cv.getContext('2d');
    const N=20,S=cv.width/N;
    let snake,dir,next,food,score,best=+(localStorage.getItem('snakeBest')||0),over,run,timer;
    document.getElementById('best').textContent=best;
    const $id=id=>document.getElementById(id);
    function reset(){snake=[{x:10,y:10},{x:9,y:10},{x:8,y:10}];dir={x:1,y:0};next=null;score=0;over=false;run=false;$id('score').textContent=0;placeFood();draw();}
    function placeFood(){do{food={x:Math.random()*N|0,y:Math.random()*N|0}}while(snake.some(s=>s.x===food.x&&s.y===food.y));}
    function step(){if(over)return;if(next){dir=next;next=null}
      const head={x:snake[0].x+dir.x,y:snake[0].y+dir.y};
      if(head.x<0||head.y<0||head.x>=N||head.y>=N||snake.some(s=>s.x===head.x&&s.y===head.y)){gameOver();return;}
      snake.unshift(head);
      if(head.x===food.x&&head.y===food.y){score++;$id('score').textContent=score;if(score>best){best=score;localStorage.setItem('snakeBest',best);$id('best').textContent=best;}placeFood();}
      else snake.pop();
      draw();}
    function draw(){ctx.fillStyle='#1e272e';ctx.fillRect(0,0,cv.width,cv.height);
      ctx.fillStyle='#e74c3c';ctx.beginPath();ctx.arc(food.x*S+S/2,food.y*S+S/2,S/2-2,0,6.283);ctx.fill();
      snake.forEach((s,i)=>{ctx.fillStyle=i===0?'#5aa8d1':'#79b6d6';ctx.beginPath();ctx.roundRect(s.x*S+1,s.y*S+1,S-2,S-2,4);ctx.fill();});
      if(over){ctx.fillStyle='rgba(0,0,0,.55)';ctx.fillRect(0,0,cv.width,cv.height);ctx.fillStyle='#fff';ctx.font='22px system-ui';ctx.textAlign='center';ctx.fillText('游戏结束，得分 '+score,cv.width/2,cv.height/2-6);ctx.font='14px system-ui';ctx.fillText('按 空格 重新开始',cv.width/2,cv.height/2+20);}
      else if(!run){ctx.fillStyle='rgba(0,0,0,.45)';ctx.fillRect(0,0,cv.width,cv.height);ctx.fillStyle='#fff';ctx.font='18px system-ui';ctx.textAlign='center';ctx.fillText('按 空格 开始',cv.width/2,cv.height/2);}}
    function gameOver(){over=true;run=false;clearInterval(timer);draw();}
    function start(){reset();run=true;timer=setInterval(step,130);}
    window.addEventListener('keydown',e=>{const k=e.key.toLowerCase();
      if(k===' '){e.preventDefault();if(over||!run)start();else run=true;return;}
      const map={arrowup:{x:0,y:-1},arrowdown:{x:0,y:1},arrowleft:{x:-1,y:0},arrowright:{x:1,y:0},w:{x:0,y:-1},s:{x:0,y:1},a:{x:-1,y:0},d:{x:1,y:0}};
      const d=map[k];if(!d)return;e.preventDefault();if(!run)start();
      if(d.x===-dir.x&&d.y===-dir.y)return;next=d;});
    document.querySelectorAll('.pad button').forEach(b=>b.addEventListener('click',()=>{
      const m={up:{x:0,y:-1},down:{x:0,y:1},left:{x:-1,y:0},right:{x:1,y:0}};const d=m[b.dataset.d];
      if(!run)start();if(d.x===-dir.x&&d.y===-dir.y)return;next=d;}));
    reset();
  </script>
</body>
</html>
