public class User {
private String name;
private String pwd;

    this.name = name;
    this.pwd = pwd;
}

public User() {
}

public String getName() {
    return name;
}

public void setName(String name) {
    this.name = name;
}

public String getPwd() {
    return pwd;
}

    this.pwd = pwd;

Study:
public class Study {
private String sno;
private String sname;
private String score;

public Study(String sno, String sname, String score) {
    this.sno = sno;
    this.sname = sname;
    this.score = score;
}

public Study() {
}

public String getSno() {
    return sno;
}

public void setSno(String sno) {
    this.sno = sno;
}

public String getSname() {
    return sname;
}

public void setSname(String sname) {
    this.sname = sname;
}

public String getScore() {
    return score;
}

public void setScore(String score) {
    this.score = score;
}

public void showInfo() {
    System.out.println("学号：" + sno + ", 姓名：" + sname + ", 成绩：" + score);
}

public static void showAllStudents(Study[] students) {

    for (int i = 0; i < students.length; i++) {
        Study student = students[i];
        student.showInfo();
    }
}

}

/for (Study student : students) {
student.showInfo(); //之前C语言里没有见过的用法，这是 Java 的增强型 for 循环（也叫 foreach 循环），专门用于遍历数组或集合。
}/

loginSuccess:
import java.util.Scanner;

public class loginSuccess {

public static boolean login(User admin) {
    while(true) {
        Scanner scanner = new Scanner(System.in);
        System.out.println("=== 学生管理系统登录 ===");
        System.out.print("请输入用户名：");
        String username = scanner.nextLine();

        System.out.print("请输入密码：");
        String pwd = scanner.nextLine();
        {
            if ("12345678".equals(pwd) && admin.getName().equals(username)) {
                System.out.println("\n=== 登录成功 ===");
                System.out.println("欢迎：" + admin.getName());
                return true;
            } else {
                System.out.println("\n登录失败！用户名或密码错误。");
                System.out.println("\n请重新登录");
            }
        }
    }
}

public static void showTopThree(Study[] students) {
    for (int i = 0; i < students.length - 1; i++) {
        for (int j = 0; j < students.length - 1 - i; j++) {
            if (Integer.parseInt(students[j].getScore()) < Integer.parseInt(students[j + 1].getScore())) {
                Study temp = students[j];
                students[j] = students[j + 1];
                students[j + 1] = temp;
            }
        }
    }

    int count = Math.min(3, students.length);
    for (int i = 0; i < count; i++) {
        System.out.println(students[i].getSname());
    }
}

}
